cmake_minimum_required(VERSION 4.2)

project(Renegade LANGUAGES C CXX RC)
option(REN_ENABLE_GAMESPY "Enable the original GameSpy SDK integration" ON)
include(cmake/Dependencies.cmake)
ren_find_dependencies()
if(NOT WIN32)
    message(FATAL_ERROR "Renegade requires Windows.")
endif()
if(NOT MSVC OR MSVC_VERSION LESS 1950)
    message(FATAL_ERROR "Renegade requires Visual Studio 2026 or newer.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Only Win32/x86 is supported by the original game and vendor ABI.")
endif()
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Build configurations" FORCE)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(cmake/LegacySources.cmake)
add_subdirectory(vendors/libbinkdec)
include(cmake/Targets.cmake)
