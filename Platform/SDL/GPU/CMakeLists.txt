set(gpu_shader_dir "${CMAKE_BINARY_DIR}/Generated/GPU")
file(MAKE_DIRECTORY "${gpu_shader_dir}")
set(gpu_shaders)
foreach(shader FixedVertex FixedFragment PresentVertex PresentFragment)
    if(shader MATCHES "Vertex$")
        set(profile vs_6_0)
    else()
        set(profile ps_6_0)
    endif()
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/Shaders/${shader}.hlsl")
    set(entry main)
    if(shader MATCHES "^Present")
        set(source "${CMAKE_CURRENT_SOURCE_DIR}/Shaders/Present.hlsl")
        string(REPLACE "Present" "" entry "${shader}")
        string(APPEND entry Main)
    endif()
    foreach(format Dxil Spirv)
        set(flags)
        if(format STREQUAL "Spirv")
            set(flags -spirv -fspv-target-env=vulkan1.0 -fvk-use-dx-layout)
        endif()
        set(binary "${gpu_shader_dir}/${shader}${format}.bin")
        set(header "${gpu_shader_dir}/${shader}${format}.h")
        add_custom_command(OUTPUT "${header}"
            COMMAND Vendor::DXC -T ${profile} -E ${entry} -O3 -Zpr ${flags} -Fo "${binary}" "${source}"
            COMMAND "${CMAKE_COMMAND}" "-DINPUT=${binary}" "-DOUTPUT=${header}" "-DSYMBOL=${shader}${format}" -P "${PROJECT_SOURCE_DIR}/CMake/EmbedShader.cmake"
            DEPENDS "${source}" "${PROJECT_SOURCE_DIR}/CMake/EmbedShader.cmake"
            VERBATIM)
        list(APPEND gpu_shaders "${header}")
    endforeach()
endforeach()
add_library(ren_graphics STATIC D3D8.cpp ${gpu_shaders})
target_include_directories(ren_graphics PUBLIC "${PROJECT_SOURCE_DIR}" PRIVATE "${gpu_shader_dir}" "${CMAKE_BINARY_DIR}/Generated/DirectX8")
target_compile_features(ren_graphics PRIVATE cxx_std_23)
target_compile_definitions(ren_graphics PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(ren_graphics PUBLIC ren_platform PRIVATE SDL3::SDL3-static)
if(NOT WIN32)
    target_sources(ren_graphics PRIVATE "${PROJECT_SOURCE_DIR}/Platform/Graphics/D3D8Guids.cpp")
endif()

add_library(ren_d3dx STATIC D3DX8.cpp TextureImages.cpp)
add_library(Vendor::d3dx8 ALIAS ren_d3dx)
target_include_directories(ren_d3dx PUBLIC "${CMAKE_BINARY_DIR}/Generated/DirectX8")
target_include_directories(ren_d3dx PRIVATE "${PROJECT_SOURCE_DIR}")
target_compile_features(ren_d3dx PRIVATE cxx_std_23)
target_compile_definitions(ren_d3dx PRIVATE NOMINMAX)
target_link_libraries(ren_d3dx PRIVATE DirectXTex SDL3_image::SDL3_image-static ren_files)
if(NOT WIN32)
    target_link_libraries(ren_d3dx PUBLIC Microsoft::DirectX-Headers)
endif()
if(WIN32)
    target_sources(ren_d3dx PRIVATE "${PROJECT_SOURCE_DIR}/Platform/Windows/GraphicsErrors.cpp")
endif()
