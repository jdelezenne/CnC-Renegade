cmake_minimum_required(VERSION 3.25)

project(Renegade LANGUAGES C CXX RC)
option(REN_ENABLE_BINK "Enable RAD Bink movie playback" ON)
option(REN_ENABLE_UMBRA "Enable the experimental legacy Umbra culler" OFF)
option(REN_ENABLE_GAMESPY "Enable the original GameSpy SDK integration" ON)
include(cmake/Dependencies.cmake)
ren_find_dependencies()
if(NOT WIN32)
    message(FATAL_ERROR "Renegade requires Windows and the 32-bit VC6 toolchain.")
endif()
if(NOT MSVC OR NOT MSVC_VERSION EQUAL 1200)
    message(FATAL_ERROR "This first build path requires Visual C++ 6.0 (_MSC_VER=1200). Modern MSVC requires a separate source port.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Only Win32/x86 is supported by the original game and vendor ABI.")
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
    message(FATAL_ERROR "Select Debug or Release using the VC6 presets.")
endif()
message(STATUS "VC6 compiler: ${CMAKE_CXX_COMPILER_VERSION}; SP5 is recommended for patch 1.037 matching")

# Keep compiler settings on targets; never request a newer C++ dialect from VC6.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase)
# NMake uses CMake's header dependency scanner, since VC6 lacks /showIncludes.
if(NOT CMAKE_GENERATOR STREQUAL "NMake Makefiles")
    message(FATAL_ERROR "Use the NMake VC6 presets: Ninja cannot track VC6 header dependencies reliably.")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(cmake/LegacySources.cmake)
include(cmake/Targets.cmake)
