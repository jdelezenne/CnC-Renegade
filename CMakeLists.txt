cmake_minimum_required(VERSION 4.2)

project(Renegade LANGUAGES C CXX RC)
option(REN_ENABLE_GAMESPY "Enable the original GameSpy SDK integration" ON)
include(CMake/Dependencies.cmake)
ren_find_dependencies()
if(NOT WIN32)
    message(FATAL_ERROR "Renegade requires Windows.")
endif()
if(NOT MSVC OR (CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND MSVC_VERSION LESS 1950))
    message(FATAL_ERROR "Renegade requires Visual Studio 2026 or newer.")
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 23)
    message(FATAL_ERROR "clang-cl requires LLVM 23 or newer. LLVM 22 miscompiles polymorphic array deletion.")
endif()
set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Build configurations" FORCE)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
# CMake maps C++23 to /std:c++latest for MSVC. Select the bounded C++23 mode.
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    set(CMAKE_CXX23_STANDARD_COMPILE_OPTION "-std:c++23preview")
    set(CMAKE_CXX23_EXTENSION_COMPILE_OPTION "-std:c++23preview")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(build_architecture x64)
else()
    set(build_architecture x86)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    set(build_compiler Clang)
else()
    set(build_compiler MSVC)
endif()
set(binary_directory "${PROJECT_SOURCE_DIR}/Binaries/Windows-${build_compiler}-${build_architecture}/$<CONFIG>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${binary_directory}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${binary_directory}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Libraries/$<CONFIG>")
include(CMake/Sources.cmake)
add_subdirectory(Vendors/LibBinkDec)
add_subdirectory(Vendors/SDL3)
add_subdirectory(Vendors/OpenALSoft)
add_subdirectory(Vendors/DrLibs)
add_subdirectory(Vendors/DXC)
add_subdirectory(Vendors/DirectXTex)
add_subdirectory(Platform)
include(CMake/Targets.cmake)
